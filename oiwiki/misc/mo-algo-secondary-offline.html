Pregled

Ponekad nailazimo na zadatke koji izgledaju vrlo pogodni za Mo algoritam (Mo's algorithm), ali pojedinačni prijelaz u njima nije \(O(1)\); tada izravna primjena Mo algoritma, čak i uz podešavanje veličine bloka, daje neispravnu složenost.

Ako se u takvom slučaju doprinos svakog prijelaza odgovoru može zapisati kao razlika (diferencirati), te prijelaze možemo rastaviti, spremiti offline i skupno obraditi drugim algoritmom.

Označimo s \(f(x, l, r)\) doprinos koji \(x\) daje s obzirom na \([l, r]\).

Kad, primjerice, trenutni interval \([l, r]\) proširujemo na \([l, r + 1]\), tražimo \(f(a_{r + 1}, l, r)\). Ako se to može diferencirati, možemo ga zapisati kao \(f(a_{r + 1}, 1, r) - f(a_{r + 1}, 1, l - 1)\), pri čemu prvi član možemo pretprocesirati za svaki \(r\), a svaki takav drugi član možemo offline spremiti na odgovarajući \(l - 1\), a zatim ih obraditi prolaskom (sweep line) od manjih prema većima. Ostali smjerovi prijelaza obrađuju se analogno.

Taj algoritam, koji povrh Mo algoritma – koji je i sam offline algoritam – još jednom offline obrađuje prijelaze, zove se dvostruko offline Mo algoritam (Mo's algorithm with secondary offline).

Razumjet ćemo ga na konkretnim zadacima.

Primjeri zadataka

Luogu P5047 [Ynoi2019 模拟赛 (probno natjecanje)] Yuno loves sqrt technology II

Zadan je niz \(a\) duljine \(n\) i \(m\) upita; svaki upit traži broj inverzija u nekom intervalu.

Ograničenja: \(1 \leq n,m \leq 10^5\), \(0 \leq a_i \leq 10^9\).

Izravnim Mo algoritmom svaki prijelaz traje barem \(O(\log n)\); uočavamo da je informacija koju pri svakom prijelazu trebamo „rang nekog broja unutar nekog intervala”, a ta se informacija s obzirom na interval može diferencirati, pa razmatramo dvostruki offline postupak.

Kad \([l, r]\) proširujemo na \([l, r + 1]\), tražimo koliko je brojeva u intervalu \([l, r]\) većih od \(a_{r + 1}\). Označimo s \(f(x, r)\) broj brojeva u intervalu \([1, r]\) većih od \(a_x\), a s \(g(x, r)\) broj brojeva u intervalu \([1, r]\) manjih od \(a_x\). Tada se promjena odgovora može zapisati kao \(f(r + 1, r) - f(r + 1, l - 1)\).

Analogno, promjena pri smanjivanju \([l, r]\) na \([l, r - 1]\) može se zapisati kao \(-f(r, r - 1) + f(r, l - 1)\); proširivanje \([l, r]\) na \([l - 1, r]\) kao \(g(l - 1, r) - g(l - 1, l - 2)\), a smanjivanje \([l, r]\) na \([l + 1, r]\) kao \(- g(l, r) + g(l, l - 1)\).

Članove \(f(x, x - 1)\) i \(g(x, x - 1)\) iz tih izraza možemo unaprijed pretprocesirati Fenwickovim stablom u \(O(n \log n)\); za ostale članove \(f(x, p)\) i \(g(x, p)\) spremimo \(x\) offline na \(l\) radi skupne obrade.

Ovdje postoji trik za uštedu memorije: primjećujemo da su pri obradi svakog upita vrijednosti \(x\) spremljene offline na \(p\) uvijek jedan neprekinuti odsječak, pa ne moramo spremati svaki pomak, nego samo odsječak za koji se vrši prilagodba. Tako memorija s ukupnog broja pomaka \(O(n\sqrt{m})\) pada na broj upita \(O(m)\).

Sad obradimo problem koji smo spremili offline: dodavanje brojeva u skup i upit za rang broja u skupu. Ukupan broj pomaka krajeva u Mo algoritmu je \(O(n\sqrt{m})\), a niz ima duljinu samo \(O(n)\), pa problem rješavamo podjelom domene vrijednosti na blokove (value-range blocking) s umetanjem u \(O(\sqrt{n})\) i upitom u \(O(1)\).

Time smo zadatak riješili u vremenskoj složenosti \(O(n \sqrt{m} + n \sqrt{n})\) i memorijskoj složenosti \(O(n + m)\).

Na kraju valja napomenuti da smo izračunali promjene odgovora pri svakom koraku, a ne same odgovore, pa konačne odgovore dobivamo tek njihovim prefiksnim sumama.

#include <algorithm>
#include <cmath>
#include <cstring>
#include <iostream>
#include <vector>

constexpr int MAX = 1e5 + 5;
constexpr int MAXX = 4e2 + 5;
constexpr int MX = 1e5;
typedef long long ll;

int n, m, sz, _sz;
int a[MAX], b[MAX], bl[MAX], st[MAXX], ed[MAXX], b1[MAXX], b2[MAX], p1[MAX],
    p2[MAX];
ll ans[MAX];

struct N {
  int l, r, id, x, y;
};

std::vector<N> v[MAX];

struct Q {
  int l, r, id;
  ll ans = 0;
} q[MAX];

bool cmp(Q a, Q b) {
  if (a.l / _sz != b.l / _sz) {
    return a.l < b.l;
  }
  return ((a.l / _sz) & 1) ? a.r < b.r : a.r > b.r;
}

void init() {
  sz = sqrt(MX);
  for (int i = 1; i <= sz; ++i) {
    st[i] = (MX / sz) * (i - 1) + 1;
    ed[i] = (MX / sz) * i;
  }
  ed[sz] = MX;
  for (int i = 1; i <= sz; ++i) {
    for (int j = st[i]; j <= ed[i]; ++j) {
      bl[j] = i;
    }
  }
}

void clear() {
  memset(b1, 0, sizeof(b1));
  memset(b2, 0, sizeof(b2));
}

void mdf(int x) {
  for (int i = bl[x]; i <= sz; ++i) {
    ++b1[i];
  }
  for (int i = x; i <= ed[bl[x]]; ++i) {
    ++b2[i];
  }
}

int qry(int x) {
  if (!x) {
    return 0;
  }
  return b1[bl[x] - 1] + b2[x];
}

int main() {
  scanf("%d%d", &n, &m);
  init();
  _sz = sqrt(m);
  for (int i = 1; i <= n; ++i) {
    scanf("%d", &a[i]);
    b[i] = a[i];
  }
  std::sort(b + 1, b + n + 1);
  int len = std::unique(b + 1, b + n + 1) - b - 1;
  for (int i = 1; i <= n; ++i) {
    a[i] = std::lower_bound(b + 1, b + len + 1, a[i]) - b;
  }
  for (int i = 1; i <= n; ++i) {
    p1[i] = qry(a[i] - 1);
    p2[i] = qry(n) - qry(a[i]);
    mdf(a[i]);
  }
  for (int i = 1; i <= m; ++i) {
    scanf("%d%d", &q[i].l, &q[i].r);
    q[i].id = i;
  }
  std::sort(q + 1, q + m + 1, cmp);
  clear();
  for (int i = 1, L = 1, R = 0; i <= m; ++i) {
    int l = q[i].l, r = q[i].r;
    if (L > l) {
      v[R].push_back({l, L - 1, i, 1, 0});
    }
    while (L > l) {
      --L;
      q[i].ans -= p1[L];
    }
    if (R < r) {
      v[L - 1].push_back({R + 1, r, i, -1, 1});
    }
    while (R < r) {
      ++R;
      q[i].ans += p2[R];
    }
    if (L < l) {
      v[R].push_back({L, l - 1, i, -1, 0});
    }
    while (L < l) {
      q[i].ans += p1[L];
      ++L;
    }
    if (R > r) {
      v[L - 1].push_back({r + 1, R, i, 1, 1});
    }
    while (R > r) {
      q[i].ans -= p2[R];
      --R;
    }
  }
  for (int i = 1; i <= n; ++i) {
    mdf(a[i]);
    for (N j : v[i]) {
      for (int k = j.l; k <= j.r; ++k) {
        if (j.y == 0) {
          q[j.id].ans += j.x * qry(a[k] - 1);
        } else {
          q[j.id].ans += j.x * (i - qry(a[k]));
        }
      }
    }
  }
  for (int i = 2; i <= m; ++i) {
    q[i].ans += q[i - 1].ans;
  }
  for (int i = 1; i <= m; ++i) {
    ans[q[i].id] = q[i].ans;
  }
  for (int i = 1; i <= m; ++i) {
    printf("%lld\n", ans[i]);
  }
}
Luogu P5501 [LnOI2019] 来者不拒，去者不追 (Tko dođe, dobrodošao; tko ode, ne zadržavamo ga)

Više puta se pita zbroj „Abbi vrijednosti” svih brojeva u intervalu \([l, r]\).

Abbi vrijednost definira se ovako: ako je \(a_i\) \(k\)-ti najmanji u intervalu upita \([l,r]\), njegova je „Abbi vrijednost” jednaka \(ka_i\).

Neka je \(f(x,r)\) zbroj brojeva u \([1,r]\) većih od \(a_x\), a \(g(x,r)\) broj brojeva u \([1,r]\) većih od \(a_x\). Tada je doprinos pri pomicanju desnog kraja udesno \(f(r,r-1)-f(r,l-1) + a_r(r-l+ 1-(g(r,r-1)-g(r,l-1)))\); ostali se smjerovi zapisuju analogno, pa ih ovdje ne navodimo.

Članovi \(f(r, r - 1)\) i \(g(r, r - 1)\) u gornjem izrazu i dalje se mogu pretprocesirati, a ostali se spremaju offline na drugi kraj i obrađuju prolaskom (sweep line). Lako se vidi da i dalje imamo problem s \(O(n)\) umetanja i \(O(n\sqrt{m})\) upita za rang, pa ga također rješavamo podjelom domene vrijednosti na blokove.

#include <algorithm>
#include <cmath>
#include <cstring>
#include <iostream>
#include <vector>

constexpr int MAX = 1e5 + 5;
constexpr int MAXX = 4e2 + 5;
constexpr int MX = 1e5;

typedef long long ll;

ll n, m, sz, _sz, sm;
ll st[MAXX], ed[MAXX], bl[MAX], a[MAX], ans[MAX], sm1[MAXX], sm3[MAXX],
    sm2[MAX], sm4[MAX], s[MAX], pre[MAX];

void init() {
  sz = sqrt(MX);
  for (int i = 1; i <= sz; ++i) {
    st[i] = (MX / sz) * (i - 1) + 1;
    ed[i] = (MX / sz) * i;
  }
  ed[sz] = MX;
  for (int i = 1; i <= sz; ++i) {
    for (int j = st[i]; j <= ed[i]; ++j) {
      bl[j] = i;
    }
  }
}

void mdf(ll x) {
  sm += x;
  for (int i = bl[x]; i <= sz; ++i) {
    ++sm1[i];
    sm3[i] += x;
  }
  for (int i = x; i <= ed[bl[x]]; ++i) {
    ++sm2[i];
    sm4[i] += x;
  }
}

ll qry(ll x) {
  if (!x) {
    return 0;
  }
  return sm1[bl[x] - 1] + sm2[x];
}

ll _qry(ll x) {
  if (!x) {
    return 0;
  }
  return sm3[bl[x] - 1] + sm4[x];
}

void clear() {
  sm = 0;
  memset(sm1, 0, sizeof(sm1));
  memset(sm2, 0, sizeof(sm2));
  memset(sm3, 0, sizeof(sm3));
  memset(sm4, 0, sizeof(sm4));
}

struct Q {
  ll l, r, id, ans;
} q[MAX];

struct N {
  ll l, r, id, x;
};

std::vector<N> v[MAX];

bool cmp(Q a, Q b) {
  if (a.l / _sz != b.l / _sz) {
    return a.l < b.l;
  } else {
    return ((a.l / _sz) & 1) ? a.r < b.r : a.r > b.r;
  }
}

int main() {
  scanf("%lld%lld", &n, &m);
  _sz = n / sqrt(m);
  init();
  for (int i = 1; i <= n; ++i) {
    scanf("%lld", &a[i]);
    s[i] = s[i - 1] + a[i];
  }
  for (int i = 1; i <= n; ++i) {
    pre[i] = qry(a[i] - 1) * a[i] + sm - _qry(a[i]);
    mdf(a[i]);
  }
  clear();
  for (int i = 1; i <= m; ++i) {
    scanf("%lld%lld", &q[i].l, &q[i].r);
    q[i].id = i;
  }
  std::sort(q + 1, q + m + 1, cmp);
  for (int i = 1, L = 1, R = 0; i <= m; ++i) {
    int l = q[i].l, r = q[i].r;
    if (L > l) {
      v[R].push_back({l, L - 1, i, 1});
    }
    while (L > l) {
      --L;
      q[i].ans -= pre[L];
    }
    if (R < r) {
      v[L - 1].push_back({R + 1, r, i, -1});
    }
    while (R < r) {
      ++R;
      q[i].ans += pre[R];
    }
    if (L < l) {
      v[R].push_back({L, l - 1, i, -1});
    }
    while (L < l) {
      q[i].ans += pre[L];
      ++L;
    }
    if (R > r) {
      v[L - 1].push_back({r + 1, R, i, 1});
    }
    while (R > r) {
      q[i].ans -= pre[R];
      --R;
    }
  }
  for (int i = 1; i <= n; ++i) {
    mdf(a[i]);
    for (N j : v[i]) {
      for (int k = j.l; k <= j.r; ++k) {
        q[j.id].ans += 1ll * j.x * (qry(a[k] - 1) * a[k] + sm - _qry(a[k]));
      }
    }
  }
  for (int i = 2; i <= m; ++i) {
    q[i].ans += q[i - 1].ans;
  }
  for (int i = 1; i <= m; ++i) {
    ans[q[i].id] = q[i].ans + s[q[i].r] - s[q[i].l - 1];
  }
  for (int i = 1; i <= m; ++i) {
    printf("%lld\n", ans[i]);
  }
  return 0;
}

Overview

Sometimes we encounter problems that look very suitable for Mo's algorithm, but whose single transition is not \(O(1)\); in that case applying Mo's algorithm directly, even with an adjusted block size, leads to an incorrect complexity.

If, in such a case, the contribution of each transition to the answer can be written as a difference, we can split these transitions apart, store them offline and process them in bulk with another algorithm.

Let \(f(x, l, r)\) denote the contribution of \(x\) with respect to \([l, r]\).

For example, when we extend the current interval \([l, r]\) to \([l, r + 1]\), what we need is \(f(a_{r + 1}, l, r)\). If it can be differenced, we can write it as \(f(a_{r + 1}, 1, r) - f(a_{r + 1}, 1, l - 1)\), where the first term can be precomputed for every \(r\), and every term of the second kind can be stored offline at the corresponding \(l - 1\) and then processed by enumerating from small to large with a sweep line. The other transition directions can be handled similarly.

This algorithm, which processes the transitions offline once more on top of Mo's algorithm, itself an offline algorithm, is called Mo's algorithm with secondary offline processing.

Let us understand it through concrete problems.

Example problems

Luogu P5047 [Ynoi2019 模拟赛 (mock contest)] Yuno loves sqrt technology II

You are given a sequence \(a\) of length \(n\) and \(m\) queries; each query asks for the number of inversions in an interval.

Constraints: \(1 \leq n,m \leq 10^5\), \(0 \leq a_i \leq 10^9\).

With plain Mo's algorithm each transition costs at least \(O(\log n)\); observing the problem, the information we need in each transition is "the rank of a number within some interval", and this information can be differenced with respect to the interval, so we consider secondary offline processing.

When we extend \([l, r]\) to \([l, r + 1]\), we need the number of elements in the interval \([l, r]\) that are greater than \(a_{r + 1}\). Let \(f(x, r)\) denote the number of elements in the interval \([1, r]\) greater than \(a_x\), and \(g(x, r)\) the number of elements in the interval \([1, r]\) smaller than \(a_x\). Then the change of the answer can be written as \(f(r + 1, r) - f(r + 1, l - 1)\).

Similarly, the change when shrinking \([l, r]\) to \([l, r - 1]\) can be written as \(-f(r, r - 1) + f(r, l - 1)\); extending \([l, r]\) to \([l - 1, r]\) can be written as \(g(l - 1, r) - g(l - 1, l - 2)\), and shrinking \([l, r]\) to \([l + 1, r]\) can be written as \(- g(l, r) + g(l, l - 1)\).

The terms \(f(x, x - 1)\) and \(g(x, x - 1)\) in these formulas can be precomputed in advance with a Fenwick tree in \(O(n \log n)\); for the remaining terms \(f(x, p)\) and \(g(x, p)\), we store \(x\) offline at \(l\) for batch processing.

There is a trick here to save memory: we notice that, when processing each query, the values \(x\) stored offline at \(p\) always form a contiguous segment, so we do not need to store every move, only the segment being adjusted. This way the memory drops from the total number of moves \(O(n\sqrt{m})\) to the number of queries \(O(m)\).

Now we handle the problem we stored offline: insert numbers into a set, and query the rank of a number in the set. The total number of endpoint moves of Mo's algorithm is \(O(n\sqrt{m})\), while the array only has length \(O(n)\), so we consider solving this problem with value-range blocking, which has \(O(\sqrt{n})\) insertion and \(O(1)\) query.

Thus we have solved this problem in \(O(n \sqrt{m} + n \sqrt{n})\) time complexity and \(O(n + m)\) space complexity.

Finally, it is worth noting that what we computed is the change of the answer at each step rather than the answer itself, so prefix sums must be taken to obtain the final answers.

Luogu P5501 [LnOI2019] 来者不拒，去者不追 (Welcome those who come, do not chase those who leave)

Multiple queries ask for the sum of the "Abbi values" of all numbers in an interval \([l, r]\).

The Abbi value is defined as follows: if \(a_i\) is the \(k\)-th smallest in the query interval \([l,r]\), then its "Abbi value" equals \(ka_i\).

Let \(f(x,r)\) be the sum of the numbers in \([1,r]\) greater than \(a_x\), and \(g(x,r)\) the number of numbers in \([1,r]\) greater than \(a_x\). Then, when we move the right endpoint to the right, the contribution produced is \(f(r,r-1)-f(r,l-1) + a_r(r-l+ 1-(g(r,r-1)-g(r,l-1)))\); the other directions can be written out analogously and are not repeated here.

The terms \(f(r, r - 1)\) and \(g(r, r - 1)\) in the formula above can still be precomputed, and the rest are stored offline at the other endpoint and processed with a sweep line. It is easy to see that what we have to handle is still a problem with \(O(n)\) insertions and \(O(n\sqrt{m})\) rank queries, so we likewise solve it with value-range blocking.

Sample code 
#include <algorithm>
#include <cmath>
#include <cstring>
#include <iostream>
#include <vector>

constexpr int MAX = 1e5 + 5;
constexpr int MAXX = 4e2 + 5;
constexpr int MX = 1e5;

typedef long long ll;

ll n, m, sz, _sz, sm;
ll st[MAXX], ed[MAXX], bl[MAX], a[MAX], ans[MAX], sm1[MAXX], sm3[MAXX],
    sm2[MAX], sm4[MAX], s[MAX], pre[MAX];

void init() {
  sz = sqrt(MX);
  for (int i = 1; i <= sz; ++i) {
    st[i] = (MX / sz) * (i - 1) + 1;
    ed[i] = (MX / sz) * i;
  }
  ed[sz] = MX;
  for (int i = 1; i <= sz; ++i) {
    for (int j = st[i]; j <= ed[i]; ++j) {
      bl[j] = i;
    }
  }
}

void mdf(ll x) {
  sm += x;
  for (int i = bl[x]; i <= sz; ++i) {
    ++sm1[i];
    sm3[i] += x;
  }
  for (int i = x; i <= ed[bl[x]]; ++i) {
    ++sm2[i];
    sm4[i] += x;
  }
}

ll qry(ll x) {
  if (!x) {
    return 0;
  }
  return sm1[bl[x] - 1] + sm2[x];
}

ll _qry(ll x) {
  if (!x) {
    return 0;
  }
  return sm3[bl[x] - 1] + sm4[x];
}

void clear() {
  sm = 0;
  memset(sm1, 0, sizeof(sm1));
  memset(sm2, 0, sizeof(sm2));
  memset(sm3, 0, sizeof(sm3));
  memset(sm4, 0, sizeof(sm4));
}

struct Q {
  ll l, r, id, ans;
} q[MAX];

struct N {
  ll l, r, id, x;
};

std::vector<N> v[MAX];

bool cmp(Q a, Q b) {
  if (a.l / _sz != b.l / _sz) {
    return a.l < b.l;
  } else {
    return ((a.l / _sz) & 1) ? a.r < b.r : a.r > b.r;
  }
}

int main() {
  scanf("%lld%lld", &n, &m);
  _sz = n / sqrt(m);
  init();
  for (int i = 1; i <= n; ++i) {
    scanf("%lld", &a[i]);
    s[i] = s[i - 1] + a[i];
  }
  for (int i = 1; i <= n; ++i) {
    pre[i] = qry(a[i] - 1) * a[i] + sm - _qry(a[i]);
    mdf(a[i]);
  }
  clear();
  for (int i = 1; i <= m; ++i) {
    scanf("%lld%lld", &q[i].l, &q[i].r);
    q[i].id = i;
  }
  std::sort(q + 1, q + m + 1, cmp);
  for (int i = 1, L = 1, R = 0; i <= m; ++i) {
    int l = q[i].l, r = q[i].r;
    if (L > l) {
      v[R].push_back({l, L - 1, i, 1});
    }
    while (L > l) {
      --L;
      q[i].ans -= pre[L];
    }
    if (R < r) {
      v[L - 1].push_back({R + 1, r, i, -1});
    }
    while (R < r) {
      ++R;
      q[i].ans += pre[R];
    }
    if (L < l) {
      v[R].push_back({L, l - 1, i, -1});
    }
    while (L < l) {
      q[i].ans += pre[L];
      ++L;
    }
    if (R > r) {
      v[L - 1].push_back({r + 1, R, i, 1});
    }
    while (R > r) {
      q[i].ans -= pre[R];
      --R;
    }
  }
  for (int i = 1; i <= n; ++i) {
    mdf(a[i]);
    for (N j : v[i]) {
      for (int k = j.l; k <= j.r; ++k) {
        q[j.id].ans += 1ll * j.x * (qry(a[k] - 1) * a[k] + sm - _qry(a[k]));
      }
    }
  }
  for (int i = 2; i <= m; ++i) {
    q[i].ans += q[i - 1].ans;
  }
  for (int i = 1; i <= m; ++i) {
    ans[q[i].id] = q[i].ans + s[q[i].r] - s[q[i].l - 1];
  }
  for (int i = 1; i <= m; ++i) {
    printf("%lld\n", ans[i]);
  }
  return 0;
}
