Dvodimenzionalni Mo algoritam (2D Mo's algorithm), kako mu i ime kaže, algoritam je u kojem se svako stanje može proširiti u četiri smjera.

U dvodimenzionalnom Mo algoritmu svako pomicanje pokazivača obrađuje brojeve u jednom retku ili jednom stupcu; konkretna je implementacija slična običnom jednodimenzionalnom Mo algoritmu, pa je ovdje ne ponavljamo. Ovdje se usredotočujemo na izbor veličine bloka.

Izbor veličine bloka

Neka je broj upita \(q\), neka su koordinate gornjeg lijevog kuta trenutne matrice \((x_1,\ y_1)\), a donjeg desnog kuta \((x_2,\ y_2)\), i neka je veličina bloka \(B\).

Tada se pokazivač \(x_1\) pomiče \(\Theta(q\cdot B)\) puta, a pokazivač \(y_2\) pomiče se \(\Theta(n^4\cdot B^{-3})\) puta.

Stoga je dovoljno uzeti \(q\cdot B=n^4\cdot B^{-3}\), tj. \(B=n\cdot q^{-\frac 14}\).

Pazite: ovako izračunani \(B\) može biti \(0\), pa taj slučaj treba posebno obraditi.

Na kraju, vremenska složenost dijela s računanjem jest \(\Theta(n^2\cdot q^{\frac 34})\); kad dodamo sortiranje upita, ukupna je vremenska složenost \(\Theta(n^2\cdot q^{\frac 34}+q\log q)\).

Primjer zadatka 1

BZOJ 2639 矩形计算 (Računanje na pravokutnicima)

Zadana je matrica \(n\times m\) čiji je svaki element cijeli broj, a zatim \(q\) upita; svaki upit traži težinu neke podmatrice. Težina matrice definira se ovako: ako se cijeli broj \(x\) u toj matrici pojavljuje \(p\) puta, on težini te matrice doprinosi \(p^2\).

Ograničenja: \(1\leq n,\ m\leq 200\), \(0\leq q\leq 10^5\), \(|\) element matrice \(| \leq 2\times 10^9\).

Ideja rješenja

Najprije diskretiziramo vrijednosti, a u dvodimenzionalnom Mo algoritmu nizom bilježimo trenutni broj pojavljivanja svakog broja.

Primjer koda 
#include <algorithm>
#include <cmath>
#include <iostream>
using namespace std;

int n, m, q, a[201][201];
long long ans[100001];
int disc[250001], cntdisc;  // za diskretizaciju

int blocklen, counts[40001];
long long now;

struct Question {
  int x1, y1, x2, y2, qid;

  bool operator<(Question tmp) const {
    if (x1 / blocklen != tmp.x1 / blocklen) return x1 < tmp.x1;
    if (y1 / blocklen != tmp.y1 / blocklen) return y1 < tmp.y1;
    if (x2 / blocklen != tmp.x2 / blocklen) return x2 < tmp.x2;
    return y2 < tmp.y2;
  }
} Q[100001];

int Qcnt;

void mo_algo_row(int id, int val, int Y1, int Y2) {
  for (int i = Y1; i <= Y2; i++)
    now -= (long long)counts[a[id][i]] * counts[a[id][i]],
        counts[a[id][i]] += val,
        now += (long long)counts[a[id][i]] * counts[a[id][i]];
}

void mo_algo_column(int id, int val, int X1, int X2) {
  for (int i = X1; i <= X2; i++)
    now -= (long long)counts[a[i][id]] * counts[a[i][id]],
        counts[a[i][id]] += val,
        now += (long long)counts[a[i][id]] * counts[a[i][id]];
}

void mo_algo() {
  blocklen = pow(n * m, 0.5) / pow(q, 0.25);
  if (blocklen < 1) blocklen = 1;
  sort(Q + 1, Q + 1 + Qcnt);

  int X1 = 1, Y1 = 1, X2 = 0, Y2 = 0;
  for (int i = 1; i <= Qcnt; i++) {
    while (X1 > Q[i].x1) mo_algo_row(--X1, 1, Y1, Y2);
    while (X2 < Q[i].x2) mo_algo_row(++X2, 1, Y1, Y2);
    while (Y1 > Q[i].y1) mo_algo_column(--Y1, 1, X1, X2);
    while (Y2 < Q[i].y2) mo_algo_column(++Y2, 1, X1, X2);
    while (X1 < Q[i].x1) mo_algo_row(X1++, -1, Y1, Y2);
    while (X2 > Q[i].x2) mo_algo_row(X2--, -1, Y1, Y2);
    while (Y1 < Q[i].y1) mo_algo_column(Y1++, -1, X1, X2);
    while (Y2 > Q[i].y2) mo_algo_column(Y2--, -1, X1, X2);
    ans[Q[i].qid] = now;
  }
}

int main() {
  cin.tie(nullptr)->sync_with_stdio(false);
  cin >> n >> m;
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++) cin >> a[i][j], disc[++cntdisc] = a[i][j];
  sort(disc + 1, disc + 1 + cntdisc);
  cntdisc = unique(disc + 1, disc + cntdisc + 1) - disc - 1;
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++)
      a[i][j] = lower_bound(disc + 1, disc + 1 + cntdisc, a[i][j]) - disc;
  cin >> q;
  for (int i = 1; i <= q; i++) {
    int x1, y1, x2, y2;
    cin >> x1 >> y1 >> x2 >> y2;
    if (x1 > x2) swap(x1, x2);
    if (y1 > y2) swap(y1, y2);
    Q[++Qcnt] = {x1, y1, x2, y2, i};
  }

  mo_algo();
  for (int i = 1; i <= q; ++i) cout << ans[i] << '\n';
  return 0;
}

Primjer zadatka 2

Luogu P1527 [国家集训队] 矩阵乘法 (Množenje matrica)

Zadana je matrica \(n\times n\) i \(q\) upita; svaki upit traži \(k\)-ti najmanji broj u nekoj podmatrici.

Ograničenja: \(1\leq n\leq 500\), \(1\leq q\leq 6\times 10^4\), \(0\leq a_{i,j}\leq 10^9\).

Najprije, kao i u prethodnom zadatku, treba diskretizirati cijelu matricu. No pazite: u ovom zadatku, osim podjele na blokove po pozicijama, treba podijeliti na blokove i domenu vrijednosti (value-range blocking) da bi se mogao izračunati odgovor.

Ovdje je za optimizaciju potrebno i sortiranje po parnosti bloka (odd-even sorting); detalje potražite u članku Obični Mo algoritam.

U ovom zadatku vremensko ograničenje nije osobito široko, pa pazite na konstantu u kodu. Izračunana veličina bloka općenito je mala: kad su \(n\) i \(q\) maksimalni, veličina bloka iznosi oko \(11\), pa se može izravno postaviti kao konstanta kako bi se uštedjelo vrijeme izvođenja.

Primjer koda 
#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int n, q, a[501][501], ans[60001];
int disc[250001], cntdisc;  // za diskretizaciju
int nn;

int blockId[501], blocklen;               // podjela na blokove
int rangeblockId[250001], rangeblocklen;  // podjela domene vrijednosti na blokove
int counts[250001], countsum[501];        // broj pojavljivanja vrijednosti i zbroj po bloku vrijednosti

struct Position {
  int x, y;
};

vector<Position> pos[250001];

struct Question {
  int x1, y1, x2, y2, k, qid;

  bool operator<(Question tmp) const {
    if (blockId[x1] != blockId[tmp.x1]) return blockId[x1] < blockId[tmp.x1];
    if (blockId[y1] != blockId[tmp.y1])
      return blockId[x1] & 1 ? y1 < tmp.y1 : y1 > tmp.y1;
    if (blockId[y2] != blockId[tmp.y2])
      return blockId[y1] & 1 ? y2 < tmp.y2 : y2 > tmp.y2;
    else
      return blockId[y2] & 1 ? x2 < tmp.x2 : x2 > tmp.x2;
  }
} Q[60001];

int Qcnt;

void mo_algo() {
  blocklen = 11;
  for (int i = 1; i <= n; ++i) blockId[i] = (i - 1) / blocklen + 1;
  rangeblocklen = n + 1;
  for (int i = 1; i <= nn; ++i) rangeblockId[i] = (i - 1) / rangeblocklen + 1;
  counts[a[1][1]] = countsum[rangeblockId[a[1][1]]] = 1;
  sort(Q + 1, Q + 1 + Qcnt);

  int L = 1, R = 1, D = 1, U = 1;
  for (int i = 1; i <= q; ++i) {
    while (R < Q[i].y2) {
      ++R;
      for (int i = U; i <= D; ++i)
        ++counts[a[i][R]], ++countsum[rangeblockId[a[i][R]]];
    }
    while (L > Q[i].y1) {
      --L;
      for (int i = U; i <= D; ++i)
        ++counts[a[i][L]], ++countsum[rangeblockId[a[i][L]]];
    }
    while (D < Q[i].x2) {
      ++D;
      for (int i = L; i <= R; ++i)
        ++counts[a[D][i]], ++countsum[rangeblockId[a[D][i]]];
    }
    while (U > Q[i].x1) {
      --U;
      for (int i = L; i <= R; ++i)
        ++counts[a[U][i]], ++countsum[rangeblockId[a[U][i]]];
    }
    while (R > Q[i].y2) {
      for (int i = U; i <= D; ++i)
        --counts[a[i][R]], --countsum[rangeblockId[a[i][R]]];
      --R;
    }
    while (L < Q[i].y1) {
      for (int i = U; i <= D; ++i)
        --counts[a[i][L]], --countsum[rangeblockId[a[i][L]]];
      ++L;
    }
    while (D > Q[i].x2) {
      for (int i = L; i <= R; ++i)
        --counts[a[D][i]], --countsum[rangeblockId[a[D][i]]];
      --D;
    }
    while (U < Q[i].x1) {
      for (int i = L; i <= R; ++i)
        --counts[a[U][i]], --countsum[rangeblockId[a[U][i]]];
      ++U;
    }
    int res = 1, cnt = 0;
    while (cnt + countsum[res] < Q[i].k && res <= rangeblockId[nn])
      cnt += countsum[res], ++res;
    res = (res - 1) * rangeblocklen + 1;
    while (cnt + counts[res] < Q[i].k && res <= nn) cnt += counts[res], ++res;
    ans[Q[i].qid] = disc[res];
  }
}

int main() {
  cin.tie(nullptr)->sync_with_stdio(false);
  cin >> n >> q;
  nn = n * n;
  for (int i = 1; i <= n; ++i)
    for (int j = 1; j <= n; ++j) {
      int x;
      cin >> x;
      a[i][j] = disc[++cntdisc] = x;
    }
  sort(disc + 1, disc + 1 + cntdisc);
  cntdisc = unique(disc + 1, disc + cntdisc + 1) - disc - 1;
  for (int i = 1; i <= n; ++i)
    for (int j = 1; j <= n; ++j)
      a[i][j] = lower_bound(disc + 1, disc + 1 + cntdisc, a[i][j]) - disc;
  for (int i = 1; i <= q; ++i) {
    int x1, y1, x2, y2, k;
    cin >> x1 >> y1 >> x2 >> y2 >> k;
    Q[++Qcnt] = {x1, y1, x2, y2, k, i};
  }

  mo_algo();
  for (int i = 1; i <= q; ++i) cout << ans[i] << '\n';
  return 0;
}

2D Mo's algorithm, as the name suggests, is Mo's algorithm in which every state can be extended in four directions.

In 2D Mo's algorithm every pointer move processes the numbers of a whole row or column; the concrete implementation is similar to ordinary one-dimensional Mo's algorithm and is not repeated here. Here we focus on the choice of the block size.

Choosing the block size

Let the number of queries be \(q\), let the top-left corner of the current matrix be \((x_1,\ y_1)\) and the bottom-right corner \((x_2,\ y_2)\), and let the block size be \(B\).

Then the pointer \(x_1\) moves \(\Theta(q\cdot B)\) times, while the pointer \(y_2\) moves \(\Theta(n^4\cdot B^{-3})\) times.

So it suffices to set \(q\cdot B=n^4\cdot B^{-3}\), i.e. \(B=n\cdot q^{-\frac 14}\).

Note that \(B\) computed this way may be \(0\), so handle this case specially.

Finally, the time complexity of the computation part is \(\Theta(n^2\cdot q^{\frac 34})\); adding the sorting of the queries, the total time complexity is \(\Theta(n^2\cdot q^{\frac 34}+q\log q)\).

Example problem 1

BZOJ 2639 矩形计算 (Rectangle Computation)

You are given an \(n\times m\) matrix whose every element is an integer, followed by \(q\) queries; each query asks for the weight of a submatrix. The weight of a matrix is defined as follows: for an integer \(x\), if it appears \(p\) times in the matrix, it contributes \(p^2\) to the weight of the matrix.

Constraints: \(1\leq n,\ m\leq 200\), \(0\leq q\leq 10^5\), \(|\) matrix element \(| \leq 2\times 10^9\).

Solution idea

First discretize the values; then, in the 2D Mo's algorithm, use an array to record the current number of occurrences of each number.

Example problem 2

Luogu P1527 [国家集训队] 矩阵乘法 (Matrix Multiplication)

You are given an \(n\times n\) matrix and \(q\) queries; each query asks for the \(k\)-th smallest number in a subrectangle.

Constraints: \(1\leq n\leq 500\), \(1\leq q\leq 6\times 10^4\), \(0\leq a_{i,j}\leq 10^9\).

First, as in the previous problem, the whole matrix needs to be discretized. Note, however, that in this problem, besides splitting the positions into blocks, we also need to split the value range into blocks (value-range blocking) in order to compute the answer.

Here we also need odd-even sorting as an optimization; for details see Ordinary Mo's algorithm.

For this problem the time limit is not very generous, so pay attention to the constant factor of the code. The computed block size is generally small: when \(n\) and \(q\) both take their maximum values the block size is about \(11\), so it can be set directly as a constant to save running time.

Sample code 
#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int n, q, a[501][501], ans[60001];
int disc[250001], cntdisc;  // for discretization
int nn;

int blockId[501], blocklen;               // block decomposition
int rangeblockId[250001], rangeblocklen;  // value-range block decomposition
int counts[250001], countsum[501];        // occurrences of each value and the total per value block

struct Position {
  int x, y;
};

vector<Position> pos[250001];

struct Question {
  int x1, y1, x2, y2, k, qid;

  bool operator<(Question tmp) const {
    if (blockId[x1] != blockId[tmp.x1]) return blockId[x1] < blockId[tmp.x1];
    if (blockId[y1] != blockId[tmp.y1])
      return blockId[x1] & 1 ? y1 < tmp.y1 : y1 > tmp.y1;
    if (blockId[y2] != blockId[tmp.y2])
      return blockId[y1] & 1 ? y2 < tmp.y2 : y2 > tmp.y2;
    else
      return blockId[y2] & 1 ? x2 < tmp.x2 : x2 > tmp.x2;
  }
} Q[60001];

int Qcnt;

void mo_algo() {
  blocklen = 11;
  for (int i = 1; i <= n; ++i) blockId[i] = (i - 1) / blocklen + 1;
  rangeblocklen = n + 1;
  for (int i = 1; i <= nn; ++i) rangeblockId[i] = (i - 1) / rangeblocklen + 1;
  counts[a[1][1]] = countsum[rangeblockId[a[1][1]]] = 1;
  sort(Q + 1, Q + 1 + Qcnt);

  int L = 1, R = 1, D = 1, U = 1;
  for (int i = 1; i <= q; ++i) {
    while (R < Q[i].y2) {
      ++R;
      for (int i = U; i <= D; ++i)
        ++counts[a[i][R]], ++countsum[rangeblockId[a[i][R]]];
    }
    while (L > Q[i].y1) {
      --L;
      for (int i = U; i <= D; ++i)
        ++counts[a[i][L]], ++countsum[rangeblockId[a[i][L]]];
    }
    while (D < Q[i].x2) {
      ++D;
      for (int i = L; i <= R; ++i)
        ++counts[a[D][i]], ++countsum[rangeblockId[a[D][i]]];
    }
    while (U > Q[i].x1) {
      --U;
      for (int i = L; i <= R; ++i)
        ++counts[a[U][i]], ++countsum[rangeblockId[a[U][i]]];
    }
    while (R > Q[i].y2) {
      for (int i = U; i <= D; ++i)
        --counts[a[i][R]], --countsum[rangeblockId[a[i][R]]];
      --R;
    }
    while (L < Q[i].y1) {
      for (int i = U; i <= D; ++i)
        --counts[a[i][L]], --countsum[rangeblockId[a[i][L]]];
      ++L;
    }
    while (D > Q[i].x2) {
      for (int i = L; i <= R; ++i)
        --counts[a[D][i]], --countsum[rangeblockId[a[D][i]]];
      --D;
    }
    while (U < Q[i].x1) {
      for (int i = L; i <= R; ++i)
        --counts[a[U][i]], --countsum[rangeblockId[a[U][i]]];
      ++U;
    }
    int res = 1, cnt = 0;
    while (cnt + countsum[res] < Q[i].k && res <= rangeblockId[nn])
      cnt += countsum[res], ++res;
    res = (res - 1) * rangeblocklen + 1;
    while (cnt + counts[res] < Q[i].k && res <= nn) cnt += counts[res], ++res;
    ans[Q[i].qid] = disc[res];
  }
}

int main() {
  cin.tie(nullptr)->sync_with_stdio(false);
  cin >> n >> q;
  nn = n * n;
  for (int i = 1; i <= n; ++i)
    for (int j = 1; j <= n; ++j) {
      int x;
      cin >> x;
      a[i][j] = disc[++cntdisc] = x;
    }
  sort(disc + 1, disc + 1 + cntdisc);
  cntdisc = unique(disc + 1, disc + cntdisc + 1) - disc - 1;
  for (int i = 1; i <= n; ++i)
    for (int j = 1; j <= n; ++j)
      a[i][j] = lower_bound(disc + 1, disc + 1 + cntdisc, a[i][j]) - disc;
  for (int i = 1; i <= q; ++i) {
    int x1, y1, x2, y2, k;
    cin >> x1 >> y1 >> x2 >> y2 >> k;
    Q[++Qcnt] = {x1, y1, x2, y2, k, i};
  }

  mo_algo();
  for (int i = 1; i <= q; ++i) cout << ans[i] << '\n';
  return 0;
}
