Uvod

U nekim zadacima pri prelasku s jednog intervala na drugi dodavanje ili brisanje može biti neizvedivo. Kad je neizvedivo samo dodavanje ili samo brisanje, problem se može riješiti Mo algoritmom s vraćanjem (rollback Mo's algorithm) u vremenu \(O(n \sqrt m)\). Osnovna ideja Mo algoritma s vraćanjem jest: ako možemo izvesti samo jednu operaciju, koristimo samo nju, a ostalo rješavamo vraćanjem (rollback).

Mo algoritam s vraćanjem dijeli se na inačicu koja koristi samo dodavanje i inačicu koja koristi samo brisanje. U nastavku opisujemo samo inačicu koja koristi samo dodavanje; inačica samo s brisanjem razlikuje se od nje tek u sitnicama implementacije, pa je ne opisujemo zasebno.

Primjer zadatka JOISC 2014 Day1 历史研究 (Povijesno istraživanje)

Zadan je niz \(A\) duljine \(n\) i \(m\) upita \((1 \leq n, m \leq 10^5)\); svaki upit traži broj s najvećom važnošću u intervalu \([L, R]\), a treba ispisati njegovu važnost. Važnost broja \(i\) definira se kao \(i\) pomnoženo s brojem pojavljivanja \(i\) u intervalu.

U ovom je zadatku ažuriranje odgovora pri dodavanju lako izvesti, ali ažuriranje pri brisanju nije. Naime, ako dodavanje utječe na odgovor, novi odgovor sigurno je važnost upravo dodanog broja; ako se pak nakon brisanja promijeni broj s najvećom važnošću u intervalu, teško je odrediti koji je broj novi broj s najvećom važnošću. Zato se ovaj zadatak teško rješava običnim Mo algoritmom.

Postupak

  • Izvorni niz podijelimo na blokove, a upite sortiramo po indeksu bloka kojem pripada lijevi kraj uzlazno kao prvom ključu i po desnom kraju uzlazno kao drugom ključu.
  • Upite obrađujemo redom:
    • Ako je blok \(B\) kojem pripada lijevi kraj upita različit od bloka kojem pripada lijevi kraj prethodnog upita, lijevi kraj Mo intervala inicijaliziramo na desni kraj bloka \(B\) plus \(1\), a desni kraj Mo intervala na desni kraj bloka \(B\);
    • Ako lijevi i desni kraj upita pripadaju istom bloku, na upit odgovaramo izravnim skeniranjem intervala;
    • Ako lijevi i desni kraj upita pripadaju različitim blokovima:
      • Ako je desni kraj upita veći od desnog kraja Mo intervala, proširujemo desni kraj dok desni kraj Mo intervala ne postane jednak desnom kraju upita;
      • Proširujemo lijevi kraj Mo intervala dok ne postane jednak lijevom kraju upita;
      • Odgovaramo na upit;
      • Poništavamo izmjene lijevog kraja Mo intervala, tj. vraćamo lijevi kraj Mo intervala na desni kraj bloka \(B\) plus \(1\).

Dokaz složenosti

Pretpostavimo da je veličina bloka u Mo algoritmu s vraćanjem \(b\):

  • Upiti kojima su lijevi i desni kraj u istom bloku mogu se izračunati u vremenu \(O(b)\);
  • Za ostale upite promotrimo upite s lijevim krajem u istom bloku: njihovi desni krajevi monotono rastu, pa je vremenska složenost pomicanja desnog kraja \(O(n)\), a pomak lijevog kraja za jedan upit ne prelazi \(b\). Budući da ima \(\frac{n}{b}\) blokova, ukupna je složenost \(O(mb+\frac{n^2}{b})\); optimalno je uzeti \(b=\frac{n}{\sqrt{m}}\), pa je vremenska složenost \(O(n\sqrt{m})\).

Implementacija

Referentni kôd 
#include <algorithm>
#include <cmath>
#include <iostream>
using namespace std;
using ll = long long;
constexpr int N = 1e5 + 5;
int n, q;
int x[N], t[N], m;

struct Query {
  int l, r, id;
} Q[N];

int pos[N], L[N], R[N], sz, tot;
int cnt[N], __cnt[N];
ll ans[N];

bool cmp(const Query& A, const Query& B) {
  if (pos[A.l] == pos[B.l]) return A.r < B.r;
  return pos[A.l] < pos[B.l];
}

void build() {
  sz = sqrt(n);
  tot = n / sz;
  for (int i = 1; i <= tot; i++) {
    L[i] = (i - 1) * sz + 1;
    R[i] = i * sz;
  }
  if (R[tot] < n) {
    ++tot;
    L[tot] = R[tot - 1] + 1;
    R[tot] = n;
  }
}

void Add(int v, ll& Ans) {
  ++cnt[v];
  Ans = max(Ans, 1LL * cnt[v] * t[v]);
}

void Del(int v) { --cnt[v]; }

int main() {
  cin.tie(nullptr)->sync_with_stdio(false);
  cin >> n >> q;
  for (int i = 1; i <= n; i++) cin >> x[i], t[++m] = x[i];
  for (int i = 1; i <= q; i++) cin >> Q[i].l >> Q[i].r, Q[i].id = i;

  build();

  // sortiranje upita
  for (int i = 1; i <= tot; i++)
    for (int j = L[i]; j <= R[i]; j++) pos[j] = i;
  sort(Q + 1, Q + 1 + q, cmp);

  // diskretizacija
  sort(t + 1, t + 1 + m);
  m = unique(t + 1, t + 1 + m) - (t + 1);
  for (int i = 1; i <= n; i++) x[i] = lower_bound(t + 1, t + 1 + m, x[i]) - t;

  int l = 1, r = 0, last_block = 0, __l;
  ll Ans = 0, tmp;
  for (int i = 1; i <= q; i++) {
    // ako su lijevi i desni kraj upita u istom bloku, odgovaramo grubom silom (skeniranjem)
    if (pos[Q[i].l] == pos[Q[i].r]) {
      for (int j = Q[i].l; j <= Q[i].r; j++) ++__cnt[x[j]];
      for (int j = Q[i].l; j <= Q[i].r; j++)
        ans[Q[i].id] = max(ans[Q[i].id], 1LL * t[x[j]] * __cnt[x[j]]);
      for (int j = Q[i].l; j <= Q[i].r; j++) --__cnt[x[j]];
      continue;
    }

    // kad dođemo do novog bloka, ponovno inicijaliziramo Mo interval
    if (pos[Q[i].l] != last_block) {
      while (r > R[pos[Q[i].l]]) Del(x[r]), --r;
      while (l < R[pos[Q[i].l]] + 1) Del(x[l]), ++l;
      Ans = 0;
      last_block = pos[Q[i].l];
    }

    // proširivanje desnog kraja
    while (r < Q[i].r) ++r, Add(x[r], Ans);
    __l = l;
    tmp = Ans;

    // proširivanje lijevog kraja
    while (__l > Q[i].l) --__l, Add(x[__l], tmp);
    ans[Q[i].id] = tmp;

    // vraćanje (rollback)
    while (__l < l) Del(x[__l]), ++__l;
  }
  for (int i = 1; i <= q; i++) cout << ans[i] << '\n';
  return 0;
}

Introduction

In some problems, when moving from one interval to another, either insertion or deletion may be impossible to implement. When only insertion is infeasible or only deletion is infeasible, the problem can be solved with rollback Mo's algorithm in \(O(n \sqrt m)\) time. The core idea of rollback Mo's algorithm is: since only one of the operations can be implemented, use only that operation and leave the rest to rollbacks.

Rollback Mo's algorithm comes in two variants, one using only insertions and one using only deletions. Below we only describe the variant that uses only insertions; the variant using only deletions differs from it only in small implementation details, so it is not described separately.

Example problem JOISC 2014 Day1 历史研究 (Historical Research)

You are given an array \(A\) of length \(n\) and \(m\) queries \((1 \leq n, m \leq 10^5)\); each query asks for the number with the largest importance in an interval \([L, R]\), and you must output its importance. The importance of a number \(i\) is defined as \(i\) multiplied by the number of occurrences of \(i\) in the interval.

In this problem, updating the answer during insertions is easy to implement, but updating it during deletions is not. If an insertion affects the answer, the new answer must be the importance of the number just inserted; however, if the number with the largest importance in the interval changes after a deletion, it is hard to determine which number is the new one with the largest importance. Therefore ordinary Mo's algorithm can hardly solve this problem.

Process

  • Split the original sequence into blocks, and sort the queries by the index of the block containing the left endpoint in ascending order as the first key and by the right endpoint in ascending order as the second key.
  • Process the queries in order:
    • If the block \(B\) containing the left endpoint of the query differs from the block containing the left endpoint of the previous query, initialize the left endpoint of the Mo interval to the right endpoint of \(B\) plus \(1\), and the right endpoint of the Mo interval to the right endpoint of \(B\);
    • If the left and right endpoints of the query belong to the same block, answer the query by scanning the interval directly;
    • If the left and right endpoints of the query belong to different blocks:
      • If the right endpoint of the query is greater than the right endpoint of the Mo interval, keep extending the right endpoint until the right endpoint of the Mo interval equals the right endpoint of the query;
      • Keep extending the left endpoint of the Mo interval until it equals the left endpoint of the query;
      • Answer the query;
      • Undo the changes to the left endpoint of the Mo interval, rolling the left endpoint of the Mo interval back to the right endpoint of \(B\) plus \(1\).

Complexity proof

Suppose the block size of rollback Mo's algorithm is \(b\):

  • Queries whose left and right endpoints lie in the same block can be computed in \(O(b)\) time;
  • For the other queries, consider the queries whose left endpoints lie in the same block: their right endpoints are monotonically increasing, so the time complexity of moving the right endpoint is \(O(n)\), while the movement of the left endpoint for a single query does not exceed \(b\). Since there are \(\frac{n}{b}\) blocks, the total complexity is \(O(mb+\frac{n^2}{b})\); taking \(b=\frac{n}{\sqrt{m}}\) is optimal, giving a time complexity of \(O(n\sqrt{m})\).

Implementation

#include <algorithm>
#include <cmath>
#include <iostream>
using namespace std;
using ll = long long;
constexpr int N = 1e5 + 5;
int n, q;
int x[N], t[N], m;

struct Query {
  int l, r, id;
} Q[N];

int pos[N], L[N], R[N], sz, tot;
int cnt[N], __cnt[N];
ll ans[N];

bool cmp(const Query& A, const Query& B) {
  if (pos[A.l] == pos[B.l]) return A.r < B.r;
  return pos[A.l] < pos[B.l];
}

void build() {
  sz = sqrt(n);
  tot = n / sz;
  for (int i = 1; i <= tot; i++) {
    L[i] = (i - 1) * sz + 1;
    R[i] = i * sz;
  }
  if (R[tot] < n) {
    ++tot;
    L[tot] = R[tot - 1] + 1;
    R[tot] = n;
  }
}

void Add(int v, ll& Ans) {
  ++cnt[v];
  Ans = max(Ans, 1LL * cnt[v] * t[v]);
}

void Del(int v) { --cnt[v]; }

int main() {
  cin.tie(nullptr)->sync_with_stdio(false);
  cin >> n >> q;
  for (int i = 1; i <= n; i++) cin >> x[i], t[++m] = x[i];
  for (int i = 1; i <= q; i++) cin >> Q[i].l >> Q[i].r, Q[i].id = i;

  build();

  // sort the queries
  for (int i = 1; i <= tot; i++)
    for (int j = L[i]; j <= R[i]; j++) pos[j] = i;
  sort(Q + 1, Q + 1 + q, cmp);

  // discretization
  sort(t + 1, t + 1 + m);
  m = unique(t + 1, t + 1 + m) - (t + 1);
  for (int i = 1; i <= n; i++) x[i] = lower_bound(t + 1, t + 1 + m, x[i]) - t;

  int l = 1, r = 0, last_block = 0, __l;
  ll Ans = 0, tmp;
  for (int i = 1; i <= q; i++) {
    // if both endpoints of the query lie in the same block, answer by brute-force scanning
    if (pos[Q[i].l] == pos[Q[i].r]) {
      for (int j = Q[i].l; j <= Q[i].r; j++) ++__cnt[x[j]];
      for (int j = Q[i].l; j <= Q[i].r; j++)
        ans[Q[i].id] = max(ans[Q[i].id], 1LL * t[x[j]] * __cnt[x[j]]);
      for (int j = Q[i].l; j <= Q[i].r; j++) --__cnt[x[j]];
      continue;
    }

    // when a new block is reached, reinitialize the Mo interval
    if (pos[Q[i].l] != last_block) {
      while (r > R[pos[Q[i].l]]) Del(x[r]), --r;
      while (l < R[pos[Q[i].l]] + 1) Del(x[l]), ++l;
      Ans = 0;
      last_block = pos[Q[i].l];
    }

    // extend the right endpoint
    while (r < Q[i].r) ++r, Add(x[r], Ans);
    __l = l;
    tmp = Ans;

    // extend the left endpoint
    while (__l > Q[i].l) --__l, Add(x[__l], tmp);
    ans[Q[i].id] = tmp;

    // roll back
    while (__l < l) Del(x[__l]), ++__l;
  }
  for (int i = 1; i <= q; i++) cout << ans[i] << '\n';
  return 0;
}
