Razlomačko programiranje (fractional programming) služi za traženje ekstrema razlomka. Formalno: zadani su \(a_i\) i \(b_i\), treba pronaći skup \(w_i\in\{0,1\}\) koji minimizira ili maksimizira

\[ \displaystyle\frac{\sum\limits_{i=1}^na_i\times w_i}{\sum\limits_{i=1}^nb_i\times w_i} \]

Jednostavnije rečeno, takvi su zadaci slični sljedećemu: svaki predmet ima dvije težine \(a\) i \(b\), a treba odabrati nekoliko predmeta tako da \(\displaystyle\frac{\sum a}{\sum b}\) bude minimalan ili maksimalan.

Zadaci razlomačkog programiranja obično imaju i neka posebna ograničenja, npr. „nazivnik je barem \(W\)”.

Rješavanje

Binarno pretraživanje

Opća metoda za zadatke razlomačkog programiranja jest binarno pretraživanje po odgovoru. Pretpostavimo da je trenutni kandidat za odgovor \(\textit{mid}\); tada bi skup \(\{w_i\}\) koji zadovoljava uvjet dao vrijednost veću ili jednaku \(\textit{mid}\). Iz tog uvjeta napišemo nejednakost i preoblikujemo je:

\[ \displaystyle \begin{aligned} &\frac{\sum a_i\times w_i}{\sum b_i\times w_i}\ge mid\\ \Longrightarrow&\sum a_i\times w_i-mid\times \sum b_i\cdot w_i\ge 0\\ \Longrightarrow&\sum w_i\times(a_i-mid\times b_i)\ge 0 \end{aligned} \]

Dakle, dovoljno je izračunati maksimum izraza na lijevoj strani nejednakosti. Ako je maksimum veći od \(0\), \(mid\) je moguć, inače nije. Glavna teškoća razlomačkog programiranja jest kako izračunati maksimum ili minimum izraza \(\displaystyle \sum w_i\times(a_i-mid\times b_i)\).

Dinkelbachov algoritam

Osnovna ideja Dinkelbachova algoritma1 jest da se u svakom koraku odgovor iz prethodnog kruga koristi kao novi ulaz \(L\) i da se iterira sve dok odgovor ne konvergira.

Primjeri zadataka

LOJ 149 01 分数规划 (01 razlomačko programiranje)

Zadano je \(n\) predmeta, svaki predmet ima dvije težine \(a\) i \(b\). Treba pronaći skup \(w_i\in\{0,1\}\) u kojem je točno \(k\) vrijednosti \(w_i\) jednako \(1\), tako da se maksimizira vrijednost \(\displaystyle\frac{\sum a_i\times w_i}{\sum b_i\times w_i}\).

Rješenje

Neka \(a_i-mid\times b_i\) bude težina \(i\)-tog predmeta; pohlepno odaberemo \(k\) predmeta s najvećim težinama. Ako je zbroj težina veći od \(0\), \(mid\) je moguć, inače nije.

Referentni kod 
#include <algorithm>
#include <cstdio>
#include <functional>
using namespace std;

constexpr int N = 100000 + 10;
constexpr double eps = 1e-6;

int n, k;
int a[N], b[N];
double c[N];

bool check(double mid) {
  double s = 0;
  for (int i = 1; i <= n; i++) c[i] = a[i] - b[i] * mid;
  // sortiraj težine od najveće do najmanje
  sort(c + 1, c + n + 1, greater<double>());
  for (int i = 1; i <= k; ++i)  // odaberi prvih k predmeta
    s += c[i];
  return s >= 0;
}

int main() {
  scanf("%d %d", &n, &k);
  for (int i = 1; i <= n; ++i) scanf("%d", &a[i]);
  for (int i = 1; i <= n; ++i) scanf("%d", &b[i]);
  double L = 0, R = 1;
  while (R - L > eps) {
    double mid = (L + R) / 2;
    if (check(mid))  // mid je moguć, odgovor je veći od mid
      L = mid;
    else  // mid nije moguć, odgovor je manji od mid
      R = mid;
  }
  printf("%.6lf\n", L);
  return 0;
}
Luogu 4377 Talent Show G

Zadano je \(n\) predmeta, svaki predmet ima dvije težine \(a\) i \(b\).

Treba odrediti skup \(w_i\in\{0,1\}\) tako da \(\displaystyle\frac{\sum w_i\times a_i}{\sum w_i\times b_i}\) bude maksimalan.

Zahtijeva se \(\displaystyle\sum w_i\times b_i \geq W\).

Rješenje

U ovom zadatku postoji dodatno ograničenje da nazivnik mora biti barem \(W\), pa se pohlepni algoritam iz prethodnog zadatka više ne može koristiti.

Možemo razmotriti 0-1 ruksak (knapsack). Neka \(b_i\) bude težina \(i\)-tog predmeta, a \(a_i-mid\times b_i\) njegova vrijednost; time se zadatak pretvara u problem ruksaka. Tada je \(dp[n][W]\) maksimum.

Tijekom DP-a zbroj težina predmeta može premašiti \(W\); u tom ga slučaju jednostavno smatramo jednakim \(W\).

Referentni kod 
#include <algorithm>
#include <cstdio>
using namespace std;

constexpr int MAXN = 250 + 10;
constexpr int MAXW = 1000 + 10;
constexpr double eps = 1e-6;

int n, W;
int w[MAXN], t[MAXN];
double f[MAXW];

bool check(double mid) {
  double s = 0;
  for (int i = 1; i <= W; i++) f[i] = -1e9;
  for (int i = 1; i <= n; i++)
    for (int j = W; j >= 0; j--) {
      int k = min(W, j + w[i]);
      f[k] = max(f[k], f[j] + t[i] - mid * w[i]);
    }
  return f[W] >= 0;
}

int main() {
  scanf("%d %d", &n, &W);
  double L = 0, R = 0;
  for (int i = 1; i <= n; ++i) {
    scanf("%d %d", &w[i], &t[i]);
    R += t[i];
  }
  while (R - L > eps) {
    double mid = (L + R) / 2;
    if (check(mid))
      L = mid;
    else
      R = mid;
  }
  printf("%d\n", (int)(L * 1000));
  return 0;
}
POJ2728 Desert King

Svaki brid ima dvije težine \(a_i\) i \(b_i\); treba pronaći razapinjuće stablo \(T\) takvo da \(\displaystyle\frac{\sum_{e\in T}a_e}{\sum_{e\in T}b_e}\) bude minimalan.

Rješenje

Neka \(a_i-mid\times b_i\) bude težina svakog brida; tada je minimalno razapinjuće stablo upravo minimum. U ovom zadatku treba pronaći minimalno razapinjuće stablo potpunog grafa, pa valja koristiti Primov algoritam.

Referentni kod 
#include <algorithm>
#include <cmath>
#include <cstdio>
using namespace std;

const int N = 1000 + 10;
const double eps = 1e-5;

int n;
double d[N][N], c[N][N], dis[N];
int x[N], y[N], z[N];
bool vis[N];

bool ok(double m) {
  double ans = 0;
  for (int i = 1; i <= n; i++) vis[i] = false;
  dis[1] = 0;
  for (int i = 2; i <= n; i++) dis[i] = 1e18;
  for (int i = 1; i <= n; i++) {
    double mn = 1e18;
    int pt = -1;
    for (int j = 1; j <= n; j++)
      if (!vis[j] && mn > dis[j]) {
        pt = j;
        mn = dis[j];
      }
    if (!~pt) break;
    vis[pt] = true;
    ans += mn;
    for (int j = 1; j <= n; j++)
      if (j != pt) dis[j] = min(dis[j], c[pt][j] - m * d[pt][j]);
  }
  return ans >= 0;
}

int main() {
  while (scanf("%d", &n) == 1) {
    if (n == 0) break;
    for (int i = 1; i <= n; i++) scanf("%d %d %d", &x[i], &y[i], &z[i]);
    for (int i = 1; i <= n; i++)
      for (int j = i + 1; j <= n; j++) {
        d[i][j] = d[j][i] =
            sqrt((x[i] - x[j]) * (x[i] - x[j]) + (y[i] - y[j]) * (y[i] - y[j]));
        c[i][j] = c[j][i] = abs(z[i] - z[j]);
      }
    double l = 0, r = 10000000;
    while (r - l > eps) {
      double m = (l + r) / 2;
      if (ok(m))
        l = m;
      else
        r = m;
    }
    printf("%.3f\n", l);
  }
  return 0;
}
[HNOI2009] 最小圈 (Minimalni ciklus)

Svaki brid ima težinu \(w\); treba pronaći ciklus \(C\) takav da \(\displaystyle\frac{\sum_{e\in C}w}{|C|}\) bude minimalan.

Rješenje

Neka \(a_i-mid\) bude težina brida; tada je ciklus s najmanjom težinom upravo minimum.

Budući da trebamo samo provjeriti je li minimum manji od \(0\), dovoljno je provjeriti postoji li u grafu negativan ciklus.

Osim toga, za ovaj zadatak postoji i algoritam složenosti \(O(nm)\); ako vas zanima, možete pročitati ovaj članak.

Referentni kod 
#include <algorithm>
#include <cstdio>
#include <tuple>
#include <vector>
using namespace std;

constexpr int N = 3000 + 10;
constexpr double eps = 1e-9;

int n, m;
double dis[N];
vector<pair<int, double>> g[N];

bool check(double mid) {  // vraća true ako postoji negativan ciklus
  bool flag = false;
  dis[0] = 0;
  for (int i = 1; i <= n; ++i) dis[i] = 1e9;
  for (int t = 1; t <= n; ++t) {
    flag = false;
    for (int u = 0; u <= n; ++u)
      for (auto vw : g[u]) {
        int v;
        double w;
        tie(v, w) = vw;
        if (dis[v] > dis[u] + w - mid) {
          dis[v] = dis[u] + w - mid;
          flag = true;
        }
      }
    if (!flag) break;
  }
  return flag;
}

int main() {
  scanf("%d %d", &n, &m);
  for (int i = 1; i <= m; ++i) {
    int u, v;
    double w;
    scanf("%d %d %lf", &u, &v, &w);
    g[u].push_back({v, w});
  }
  for (int i = 1; i <= n; i++) g[0].push_back({i, 0});
  double L = -1e7, R = 1e7;
  while (R - L > eps) {
    double mid = (L + R) / 2;
    if (check(mid))
      R = mid;
    else
      L = mid;
  }
  printf("%.8lf\n", L);
  return 0;
}

Fractional programming is used to find the extremum of a fraction. Formally: given \(a_i\) and \(b_i\), find a set of \(w_i\in\{0,1\}\) that minimizes or maximizes

\[ \displaystyle\frac{\sum\limits_{i=1}^na_i\times w_i}{\sum\limits_{i=1}^nb_i\times w_i} \]

Informally, such problems look like this: each item has two weights \(a\) and \(b\), and we have to choose some items so that \(\displaystyle\frac{\sum a}{\sum b}\) is minimized or maximized.

Fractional programming problems usually also have some special constraints, such as "the denominator is at least \(W\)".

Solving

Binary search

The general method for fractional programming problems is binary search on the answer. Suppose the current candidate answer is \(\textit{mid}\); then a set \(\{w_i\}\) satisfying the condition would give a value greater than or equal to \(\textit{mid}\). From this condition we write an inequality and transform it:

\[ \displaystyle \begin{aligned} &\frac{\sum a_i\times w_i}{\sum b_i\times w_i}\ge mid\\ \Longrightarrow&\sum a_i\times w_i-mid\times \sum b_i\cdot w_i\ge 0\\ \Longrightarrow&\sum w_i\times(a_i-mid\times b_i)\ge 0 \end{aligned} \]

So it suffices to compute the maximum of the expression on the left-hand side of the inequality. If the maximum is greater than \(0\), \(mid\) is feasible, otherwise it is not. The main difficulty of fractional programming is how to compute the maximum or minimum of \(\displaystyle \sum w_i\times(a_i-mid\times b_i)\).

Dinkelbach's algorithm

The rough idea of Dinkelbach's algorithm1 is to use the answer of the previous round as the new input \(L\) each time, iterating until the answer converges.

Example problems

LOJ 149 01 分数规划 (0-1 fractional programming)

There are \(n\) items, each with two weights \(a\) and \(b\). Find a set of \(w_i\in\{0,1\}\) in which exactly \(k\) of the \(w_i\) are \(1\), maximizing the value of \(\displaystyle\frac{\sum a_i\times w_i}{\sum b_i\times w_i}\).

Solution

Take \(a_i-mid\times b_i\) as the weight of the \(i\)-th item and greedily pick the \(k\) items with the largest weights. If the sum of the weights is greater than \(0\), \(mid\) is feasible, otherwise it is not.

Reference code 
#include <algorithm>
#include <cstdio>
#include <functional>
using namespace std;

constexpr int N = 100000 + 10;
constexpr double eps = 1e-6;

int n, k;
int a[N], b[N];
double c[N];

bool check(double mid) {
  double s = 0;
  for (int i = 1; i <= n; i++) c[i] = a[i] - b[i] * mid;
  // sort the weights in decreasing order
  sort(c + 1, c + n + 1, greater<double>());
  for (int i = 1; i <= k; ++i)  // pick the first k items
    s += c[i];
  return s >= 0;
}

int main() {
  scanf("%d %d", &n, &k);
  for (int i = 1; i <= n; ++i) scanf("%d", &a[i]);
  for (int i = 1; i <= n; ++i) scanf("%d", &b[i]);
  double L = 0, R = 1;
  while (R - L > eps) {
    double mid = (L + R) / 2;
    if (check(mid))  // mid is feasible, the answer is larger than mid
      L = mid;
    else  // mid is infeasible, the answer is smaller than mid
      R = mid;
  }
  printf("%.6lf\n", L);
  return 0;
}
Luogu 4377 Talent Show G

There are \(n\) items, each with two weights \(a\) and \(b\).

You have to determine a set of \(w_i\in\{0,1\}\) such that \(\displaystyle\frac{\sum w_i\times a_i}{\sum w_i\times b_i}\) is maximized.

It is required that \(\displaystyle\sum w_i\times b_i \geq W\).

Solution

This problem has the additional constraint that the denominator must be at least \(W\), so the greedy algorithm of the previous problem can no longer be used.

We can consider a 0-1 knapsack. Take \(b_i\) as the weight of the \(i\)-th item and \(a_i-mid\times b_i\) as its value; the problem then turns into a knapsack problem. Then \(dp[n][W]\) is the maximum.

During the DP the total weight of the items may exceed \(W\); in that case we simply treat it as \(W\).

Reference code 
#include <algorithm>
#include <cstdio>
using namespace std;

constexpr int MAXN = 250 + 10;
constexpr int MAXW = 1000 + 10;
constexpr double eps = 1e-6;

int n, W;
int w[MAXN], t[MAXN];
double f[MAXW];

bool check(double mid) {
  double s = 0;
  for (int i = 1; i <= W; i++) f[i] = -1e9;
  for (int i = 1; i <= n; i++)
    for (int j = W; j >= 0; j--) {
      int k = min(W, j + w[i]);
      f[k] = max(f[k], f[j] + t[i] - mid * w[i]);
    }
  return f[W] >= 0;
}

int main() {
  scanf("%d %d", &n, &W);
  double L = 0, R = 0;
  for (int i = 1; i <= n; ++i) {
    scanf("%d %d", &w[i], &t[i]);
    R += t[i];
  }
  while (R - L > eps) {
    double mid = (L + R) / 2;
    if (check(mid))
      L = mid;
    else
      R = mid;
  }
  printf("%d\n", (int)(L * 1000));
  return 0;
}
POJ2728 Desert King

Each edge has two weights \(a_i\) and \(b_i\); find a spanning tree \(T\) such that \(\displaystyle\frac{\sum_{e\in T}a_e}{\sum_{e\in T}b_e}\) is minimized.

Solution

Take \(a_i-mid\times b_i\) as the weight of each edge; then the minimum spanning tree is exactly the minimum. In this problem we need the minimum spanning tree of a complete graph, so Prim's algorithm should be used.

Reference code 
#include <algorithm>
#include <cmath>
#include <cstdio>
using namespace std;

const int N = 1000 + 10;
const double eps = 1e-5;

int n;
double d[N][N], c[N][N], dis[N];
int x[N], y[N], z[N];
bool vis[N];

bool ok(double m) {
  double ans = 0;
  for (int i = 1; i <= n; i++) vis[i] = false;
  dis[1] = 0;
  for (int i = 2; i <= n; i++) dis[i] = 1e18;
  for (int i = 1; i <= n; i++) {
    double mn = 1e18;
    int pt = -1;
    for (int j = 1; j <= n; j++)
      if (!vis[j] && mn > dis[j]) {
        pt = j;
        mn = dis[j];
      }
    if (!~pt) break;
    vis[pt] = true;
    ans += mn;
    for (int j = 1; j <= n; j++)
      if (j != pt) dis[j] = min(dis[j], c[pt][j] - m * d[pt][j]);
  }
  return ans >= 0;
}

int main() {
  while (scanf("%d", &n) == 1) {
    if (n == 0) break;
    for (int i = 1; i <= n; i++) scanf("%d %d %d", &x[i], &y[i], &z[i]);
    for (int i = 1; i <= n; i++)
      for (int j = i + 1; j <= n; j++) {
        d[i][j] = d[j][i] =
            sqrt((x[i] - x[j]) * (x[i] - x[j]) + (y[i] - y[j]) * (y[i] - y[j]));
        c[i][j] = c[j][i] = abs(z[i] - z[j]);
      }
    double l = 0, r = 10000000;
    while (r - l > eps) {
      double m = (l + r) / 2;
      if (ok(m))
        l = m;
      else
        r = m;
    }
    printf("%.3f\n", l);
  }
  return 0;
}
[HNOI2009] 最小圈 (Minimum cycle)

Each edge has weight \(w\); find a cycle \(C\) such that \(\displaystyle\frac{\sum_{e\in C}w}{|C|}\) is minimized.

Solution

Take \(a_i-mid\) as the edge weight; then the cycle with the smallest weight is exactly the minimum.

Since we only need to check whether the minimum is less than \(0\), it suffices to check whether the graph contains a negative cycle.

In addition, there is an \(O(nm)\) algorithm for this problem; if you are interested, you can read this article.

Reference code 
#include <algorithm>
#include <cstdio>
#include <tuple>
#include <vector>
using namespace std;

constexpr int N = 3000 + 10;
constexpr double eps = 1e-9;

int n, m;
double dis[N];
vector<pair<int, double>> g[N];

bool check(double mid) {  // returns true if there is a negative cycle
  bool flag = false;
  dis[0] = 0;
  for (int i = 1; i <= n; ++i) dis[i] = 1e9;
  for (int t = 1; t <= n; ++t) {
    flag = false;
    for (int u = 0; u <= n; ++u)
      for (auto vw : g[u]) {
        int v;
        double w;
        tie(v, w) = vw;
        if (dis[v] > dis[u] + w - mid) {
          dis[v] = dis[u] + w - mid;
          flag = true;
        }
      }
    if (!flag) break;
  }
  return flag;
}

int main() {
  scanf("%d %d", &n, &m);
  for (int i = 1; i <= m; ++i) {
    int u, v;
    double w;
    scanf("%d %d %lf", &u, &v, &w);
    g[u].push_back({v, w});
  }
  for (int i = 1; i <= n; i++) g[0].push_back({i, 0});
  double L = -1e7, R = 1e7;
  while (R - L > eps) {
    double mid = (L + R) / 2;
    if (check(mid))
      R = mid;
    else
      L = mid;
  }
  printf("%.8lf\n", L);
  return 0;
}
